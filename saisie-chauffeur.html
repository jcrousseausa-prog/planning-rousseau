async function chargerRecapSemaine(){ const employeeId = document.getElementById('selChauffeur').value; const date = document.getElementById('selDate').value; if (!employeeId || !date) return; try { const res = await fetch(POINTAGE_SCRIPT_URL, { method:'POST', headers:{'Content-Type':'text/plain'}, body: JSON.stringify({action:'getRecapSemaineChauffeur', employeeId, date}) }); const r = await res.json(); if (!r.ok) return; const jours = ['Lun','Mar','Mer','Jeu','Ven','Sam','Dim']; const lundi = new Date(r.lundi); let totalSem = 0; let html = ''; for (let i=0;i<7;i++){ const d = new Date(lundi.getTime()); d.setDate(lundi.getDate()+i); const key = d.toISOString().slice(0,10); const h = r.parJour[key] || 0; totalSem += h; const estAujourdhui = key === date; html += `
${jours[i]}
${h>0?fmtHeures(h):'—'}
`; } document.getElementById('detailSemaine').innerHTML = html; document.getElementById('totalSemaine').textContent = fmtHeures(totalSem); } catch(e) { console.warn(e); } }
